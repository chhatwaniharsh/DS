#8
num <- as.integer(readline("Enter a number: "))
repeat {
  sum <- 0
  temp <- num
  
  while(temp > 0) {
    digit <- temp %% 10
    sum <- sum + digit
    temp <- temp %/% 10
  }
  
  if(sum < 10) {
    break
  }
  num <- sum
}
cat("Single digit sum is:", sum)
